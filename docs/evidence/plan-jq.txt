# Независимая проверка terraform/plan.json (без агента)

## 1. Все изменения и actions
$ jq -r '.resource_changes[] | "\(.address)\t\(.change.actions|join(","))"' terraform/plan.json
aws_db_instance.launch_manifest	delete,create
aws_db_parameter_group.legacy	delete
aws_ecs_task_definition.api	create,delete
aws_iam_policy.api	update
aws_security_group.api	update
aws_cloudwatch_log_group.api	create
aws_s3_bucket.artifacts	no-op
aws_ecs_service.api	update
aws_ssm_parameter.app_env	update

## 2. Изменения с delete (replace = delete+create)
$ jq -r '.resource_changes[] | select(.change.actions | index("delete")) | .address' terraform/plan.json
aws_db_instance.launch_manifest
aws_db_parameter_group.legacy
aws_ecs_task_definition.api

## 6. Верхнеуровневые ключи плана

## 7. Ключи change (полнота)
$ jq -r '[.resource_changes[].change | keys[]] | unique' terraform/plan.json
[
  "actions",
  "after",
  "before"
]
